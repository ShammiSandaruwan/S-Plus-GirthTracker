function doPost(e) {
  try {
    const parsed = JSON.parse(e.postData.contents);
    const action = parsed.action;

    if (action === "request_access") {
      return handleRequestAccess(parsed);
    } else if (action === "check_access") {
      return handleCheckAccess(parsed);
    } else if (action === "sync_measurements") {
      return handleSyncMeasurements(parsed);
    } else if (action === "admin_verify_totp") {
      return handleAdminVerifyTotp(parsed);
    } else if (action === "admin_list_estates") {
      return handleAdminListEstates(parsed);
    } else if (action === "admin_fetch_measurements") {
      return handleAdminFetchMeasurements(parsed);
    } else if (action === "admin_list_devices") {
      return handleAdminListDevices(parsed);
    } else if (action === "admin_revoke_device") {
      return handleAdminRevokeDevice(parsed);
    } else if (action === "delete_measurement") {
      return handleDeleteMeasurement(parsed);
    }

    return respondJson({ success: false, error: "Unknown action" });
  } catch (error) {
    return respondJson({ success: false, error: error.toString() });
  }
}

function doGet(e) {
  const action = e.parameter.action;
  const requestId = e.parameter.requestId;
  const adminKey = e.parameter.adminKey;

  if (adminKey !== ADMIN_APPROVAL_KEY) {
    return ContentService.createTextOutput("Invalid admin key.");
  }

  if (!requestId) {
    return ContentService.createTextOutput("Request not found.");
  }

  try {
    if (action === "approve") {
      return handleApprove(requestId);
    } else if (action === "deny") {
      return handleDeny(requestId);
    }
    return ContentService.createTextOutput("Unknown action.");
  } catch (error) {
    return ContentService.createTextOutput("Error: " + error.toString());
  }
}

// Request Access
// ----------------------------------------------------
function handleRequestAccess(parsed) {
  const { estate, operatorName, deviceId, location, userAgent, appVersion } = parsed;
  const requestId = "REQ-" + Utilities.getUuid();
  const deviceIdHash = computeHash(deviceId);

  const sheet = getOrCreateSheet("AccessRequests", [
    "requestId", "createdAt", "estate", "operatorName", "deviceIdHash",
    "latitude", "longitude", "gpsAccuracy", "gpsStatus", "googleMapLink",
    "userAgent", "appVersion", "status", "approvedAt", "deniedAt", "adminAction", "notes"
  ]);

  sheet.appendRow([
    requestId,
    new Date().toISOString(),
    estate,
    operatorName,
    deviceIdHash,
    location?.latitude || "",
    location?.longitude || "",
    location?.accuracy || "",
    location?.status || "",
    location?.googleMapLink || "",
    userAgent || "",
    appVersion || "",
    "pending",
    "",
    "",
    "",
    ""
  ]);

  const safeEstate = safeTelegramText_(estate);
  const safeOperator = safeTelegramText_(operatorName);
  const safeGpsStatus = safeTelegramText_(location?.status || 'unknown');

  const text = `New GirthTracker Access Request\n\n` +
               `Estate: ${safeEstate}\n` +
               `Operator Name: ${safeOperator}\n` +
               `GPS Status: ${safeGpsStatus}\n` +
               `Requested At: ${new Date().toISOString()}\n\n` +
               `Approve or deny this device.`;

  const inlineKeyboard = [
    [
      { text: "✅ Approve", url: `${WEB_APP_URL}?action=approve&requestId=${requestId}&adminKey=${ADMIN_APPROVAL_KEY}` },
      { text: "❌ Deny", url: `${WEB_APP_URL}?action=deny&requestId=${requestId}&adminKey=${ADMIN_APPROVAL_KEY}` }
    ]
  ];

  if (location?.googleMapLink) {
    inlineKeyboard.push([
      { text: "📍 Open Location", url: location.googleMapLink }
    ]);
  }

  const telegramResult = sendTelegramNotification_(text, inlineKeyboard);
  let telegramStatus = telegramResult.status;

  if (telegramStatus === "failed") {
    const lastRow = sheet.getLastRow();
    sheet.getRange(lastRow, 17).setValue(telegramResult.error || "Telegram send failed.");
  }

  return respondJson({
    success: true,
    requestId: requestId,
    status: "pending",
    message: "Access request submitted.",
    telegramStatus: telegramStatus
  });
}

// ----------------------------------------------------
// Check Access
// ----------------------------------------------------
function handleCheckAccess(parsed) {
  const { requestId, deviceId } = parsed;
  const deviceIdHash = computeHash(deviceId);

  const reqSheet = getOrCreateSheet("AccessRequests");
  const data = reqSheet.getDataRange().getValues();
  const headers = data[0];
  
  let status = "not_found";
  let approvedAt = "";
  let estate = "";
  let operatorName = "";

  for (let i = 1; i < data.length; i++) {
    if (data[i][headers.indexOf("requestId")] === requestId && data[i][headers.indexOf("deviceIdHash")] === deviceIdHash) {
      status = data[i][headers.indexOf("status")];
      approvedAt = data[i][headers.indexOf("approvedAt")];
      estate = data[i][headers.indexOf("estate")];
      operatorName = data[i][headers.indexOf("operatorName")];
      break;
    }
  }

  if (status === "pending") {
    return respondJson({ success: true, status: "pending" });
  } else if (status === "denied") {
    return respondJson({ success: true, status: "denied", message: "Access request denied." });
  } else if (status === "approved") {
    const deviceToken = getDeviceToken(deviceIdHash);
    if (!deviceToken) {
       return respondJson({ success: false, error: "Token not found despite approval." });
    }
    return respondJson({
      success: true,
      status: "approved",
      deviceToken: deviceToken,
      estate: estate,
      operatorName: operatorName,
      approvedAt: approvedAt,
      expiresAt: ""
    });
  }

  return respondJson({ success: false, error: "Request not found." });
}

// Sync Measurements
// ----------------------------------------------------
function handleSyncMeasurements(parsed) {
  const { deviceId, deviceToken, estate, operatorName, measurements } = parsed;

  const authResult = validateDevice(deviceId, deviceToken, estate, operatorName);
  if (!authResult.success) {
    return respondJson({
      success: false,
      errorType: authResult.errorType,
      error: authResult.error
    });
  }

  if (!Array.isArray(measurements) || measurements.length === 0) {
    return respondJson({ success: true, message: "Empty batch" });
  }

  const sheetId = ESTATE_SHEET_MAP[estate];
  let sheet;
  if (sheetId) {
    sheet = SpreadsheetApp.openById(sheetId).getSheets()[0];
  } else {
    sheet = getOrCreateSheet("Measurements_" + estate);
  }

  ensureMeasurementSheetSchema_(sheet);

  const deviceIdHash = computeHash(deviceId);
  
  const rowsToAppend = [];

  // Server-side deduplication: exact Measurement ID match means this row was already synced.
  // Legacy fallback to treeNo|timestamp covers migration edge cases.
  const existingData = sheet.getDataRange().getValues();
  const existingHeaders = existingData[0];
  let colMap;
  try {
    colMap = getRequiredMeasurementColumnMap_(existingHeaders);
  } catch (e) {
    return respondJson({ success: false, error: e.message });
  }

  const treeCol = colMap["Tree Number"];
  const dateCol = colMap["Date"];
  const midCol = colMap["Measurement ID"];
  const existingMids = new Set();
  const existingLegacyKeys = new Set();
  for (let i = 1; i < existingData.length; i++) {
    // Index by measurement id when present
    if (midCol !== -1 && existingData[i][midCol]) {
      existingMids.add(String(existingData[i][midCol]));
    }
    // Always build legacy key index for backward compat
    const legacyKey = String(existingData[i][treeCol]) + "|" + String(existingData[i][dateCol]);
    existingLegacyKeys.add(legacyKey);
  }

  const syncedIds = [];
  const skippedIds = [];

  measurements.forEach(m => {
    const formattedDate = formatSriLankaDateTime_(m.timestamp);
    const mid = m.id != null ? String(m.id) : "";
    const legacyKey = String(valueOrBlank_(m.treeNo)) + "|" + formattedDate;

    // Dedup: exact Measurement ID match means this row was already synced
    if (mid && existingMids.has(mid)) {
      skippedIds.push(m.id);
      return;
    }
    // Also check legacy key — covers migration edge case where the same measurement
    // was synced before Measurement ID existed (row has matching treeNo|timestamp but no mid)
    if (existingLegacyKeys.has(legacyKey)) {
      skippedIds.push(m.id);
      return;
    }
    if (mid) existingMids.add(mid);
    existingLegacyKeys.add(legacyKey);
    syncedIds.push(m.id);
    rowsToAppend.push([
      m.estate || estate,
      m.division || "",
      m.fieldNo || "",
      valueOrBlank_(m.extent),
      valueOrBlank_(m.treeNo),
      valueOrBlank_(m.caliperReading),
      valueOrBlank_(m.girth),
      formattedDate,
      m.recommendationText || "",
      m.abnormalFlag ? "Yes" : "No",
      m.abnormalReason || "",
      valueOrBlank_(m.latitude),
      valueOrBlank_(m.longitude),
      valueOrBlank_(m.gpsAccuracy),
      m.googleMapLink || createGoogleMapLink_(m.latitude, m.longitude),
      mid
    ]);
  });

  if (rowsToAppend.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, rowsToAppend.length, rowsToAppend[0].length).setValues(rowsToAppend);
  }

  updateLastSeen(deviceIdHash);

  return respondJson({
    success: true,
    syncedIds: syncedIds.concat(skippedIds),
    failed: []
  });
}

// ----------------------------------------------------
// Delete Measurement (Undo)
// ----------------------------------------------------
function handleDeleteMeasurement(parsed) {
  const { deviceId, deviceToken, estate, measurementId, treeNo, timestamp } = parsed;

  const authResult = validateDevice(deviceId, deviceToken, estate);
  if (!authResult.success) {
    return respondJson({
      success: false,
      errorType: authResult.errorType,
      error: authResult.error
    });
  }

  const sheetId = ESTATE_SHEET_MAP[estate];
  if (!sheetId) {
    return respondJson({ success: false, error: "Unknown estate." });
  }

  try {
    const sheet = SpreadsheetApp.openById(sheetId).getSheets()[0];
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    
    let colMap;
    try {
      colMap = getRequiredMeasurementColumnMap_(headers);
    } catch (e) {
      return respondJson({ success: false, error: e.message });
    }
    const midCol = colMap["Measurement ID"];
    const treeCol = colMap["Tree Number"];
    const dateCol = colMap["Date"];

    // Primary: delete by unique Measurement ID (exact match)
    if (measurementId != null && midCol !== -1) {
      for (let i = data.length - 1; i >= 1; i--) {
        if (String(data[i][midCol]) === String(measurementId)) {
          sheet.deleteRow(i + 1);
          return respondJson({ success: true });
        }
      }
    }

    // Fallback: legacy rows without Measurement ID — match by treeNo + timestamp
    if (treeNo != null && timestamp) {
      const formattedDate = formatSriLankaDateTime_(timestamp);
      for (let i = data.length - 1; i >= 1; i--) {
        if (String(data[i][treeCol]) === String(treeNo) && String(data[i][dateCol]) === formattedDate) {
          sheet.deleteRow(i + 1);
          return respondJson({ success: true });
        }
      }
    }

    return respondJson({ success: false, error: "Row not found." });
  } catch (e) {
    return respondJson({ success: false, error: "Error deleting row: " + e.toString() });
  }
}

function handleApprove(requestId) {
  const reqSheet = getOrCreateSheet("AccessRequests");
  const reqData = reqSheet.getDataRange().getValues();
  const headers = reqData[0];
  
  let rowIndex = -1;
  let rowData = null;

  for (let i = 1; i < reqData.length; i++) {
    if (reqData[i][headers.indexOf("requestId")] === requestId) {
      rowIndex = i + 1;
      rowData = reqData[i];
      break;
    }
  }

  if (rowIndex === -1) return ContentService.createTextOutput("Request not found.");
  
  const status = rowData[headers.indexOf("status")];
  if (status === "approved") return ContentService.createTextOutput("This request has already been approved.");

  const deviceIdHash = rowData[headers.indexOf("deviceIdHash")];
  const estate = rowData[headers.indexOf("estate")];
  const operatorName = rowData[headers.indexOf("operatorName")];
  
  const deviceToken = "TOK-" + Utilities.getUuid();
  const tokenHash = computeHash(deviceToken);
  const approvedAt = new Date().toISOString();

  // Update AccessRequests
  reqSheet.getRange(rowIndex, headers.indexOf("status") + 1).setValue("approved");
  reqSheet.getRange(rowIndex, headers.indexOf("approvedAt") + 1).setValue(approvedAt);
  reqSheet.getRange(rowIndex, headers.indexOf("adminAction") + 1).setValue("approved via telegram");

  // Save to ApprovedDevices
  const appSheet = getOrCreateSheet("ApprovedDevices", [
    "deviceIdHash", "tokenHash", "estate", "operatorName", "approvedAt", "expiresAt", "revoked", "lastSeenAt", "lastSyncAt", "notes"
  ]);
  
  // Check if device already exists, update or append
  const appData = appSheet.getDataRange().getValues();
  let existingAppRow = -1;
  for(let i=1; i < appData.length; i++){
     if(appData[i][0] === deviceIdHash){
       existingAppRow = i + 1;
       break;
     }
  }

  if (existingAppRow > -1) {
      appSheet.getRange(existingAppRow, 2).setValue(tokenHash); // tokenHash
      appSheet.getRange(existingAppRow, 3).setValue(estate); // estate
      appSheet.getRange(existingAppRow, 4).setValue(operatorName); // operatorName
      appSheet.getRange(existingAppRow, 5).setValue(approvedAt); // approvedAt
      appSheet.getRange(existingAppRow, 7).setValue(false); // revoked
  } else {
      appSheet.appendRow([
        deviceIdHash, tokenHash, estate, operatorName, approvedAt, "", false, "", "", ""
      ]);
  }

  // We need to pass the raw token back to the device, so we temporarily store it in a Cache to be retrieved by the check_access call
  CacheService.getScriptCache().put("token_" + deviceIdHash, deviceToken, 21600); // 6 hours

  return ContentService.createTextOutput("Access approved successfully.");
}

function handleDeny(requestId) {
  const reqSheet = getOrCreateSheet("AccessRequests");
  const reqData = reqSheet.getDataRange().getValues();
  const headers = reqData[0];
  
  let rowIndex = -1;

  for (let i = 1; i < reqData.length; i++) {
    if (reqData[i][headers.indexOf("requestId")] === requestId) {
      rowIndex = i + 1;
      break;
    }
  }

  if (rowIndex === -1) return ContentService.createTextOutput("Request not found.");

  reqSheet.getRange(rowIndex, headers.indexOf("status") + 1).setValue("denied");
  reqSheet.getRange(rowIndex, headers.indexOf("deniedAt") + 1).setValue(new Date().toISOString());
  reqSheet.getRange(rowIndex, headers.indexOf("adminAction") + 1).setValue("denied via telegram");

  return ContentService.createTextOutput("Access request denied.");
}

// ----------------------------------------------------
// Utilities
// ----------------------------------------------------
function computeHash(input) {
  if (!input) return "";
  const rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input);
  let txtHash = '';
  for (let i = 0; i < rawHash.length; i++) {
    let hashVal = rawHash[i];
    if (hashVal < 0) hashVal += 256;
    if (hashVal.toString(16).length == 1) txtHash += '0';
    txtHash += hashVal.toString(16);
  }
  return txtHash;
}

function respondJson(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(sheetName, headers = []) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    if (headers.length > 0) {
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold");
    }
  }
  return sheet;
}

/**
 * Safe, idempotent schema migration for measurement sheets.
 * - If the sheet is empty, writes the full MEASUREMENT_HEADERS row.
 * - If the sheet already has legacy 15-column headers, appends "Measurement ID"
 *   as column 16 without disturbing existing data rows (they get blank in col 16).
 * - If "Measurement ID" already exists, does nothing.
 * - Validates that known required headers are present; logs a warning if not.
 * - Running repeatedly is safe — no duplicate columns, no data corruption.
 */
function ensureMeasurementSheetSchema_(sheet) {
  const lastCol = sheet.getLastColumn();

  // Empty sheet — write full header row
  if (lastCol === 0) {
    sheet.getRange(1, 1, 1, MEASUREMENT_HEADERS.length).setValues([MEASUREMENT_HEADERS]);
    sheet.getRange(1, 1, 1, MEASUREMENT_HEADERS.length).setFontWeight("bold");
    return;
  }

  const currentHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

  // Validate that expected core headers are present (sanity check)
  const requiredCore = ["Estate", "Tree Number", "Date", "Girth"];
  for (let i = 0; i < requiredCore.length; i++) {
    if (currentHeaders.indexOf(requiredCore[i]) === -1) {
      Logger.log("WARNING: Measurement sheet is missing expected header '" + requiredCore[i] + "'. Schema may be malformed.");
    }
  }

  // Check if "Measurement ID" column already exists
  if (currentHeaders.indexOf("Measurement ID") !== -1) {
    return; // Already migrated, nothing to do
  }

  // Append the new header in the next available column (preserves existing alignment)
  const newColIndex = lastCol + 1;
  sheet.getRange(1, newColIndex).setValue("Measurement ID");
  sheet.getRange(1, newColIndex).setFontWeight("bold");
  // Existing data rows naturally have blank cells in this new column
}

/**
 * Strict validation: confirms all required measurement columns exist in the header row.
 * Returns a map of column names to indexes. Throws an error if any required column is missing.
 */
function getRequiredMeasurementColumnMap_(headers) {
  if (!headers || headers.length === 0) {
    throw new Error("Measurement sheet headers are empty.");
  }
  
  const requiredHeaders = [
    "Estate", "Division", "Field No", "Extent (Ha)", "Tree Number", 
    "Caliper Reading", "Girth", "Date", "Tapping Recommendation", 
    "Abnormal Flag", "Abnormal Reason", "Latitude", "Longitude", 
    "GPS Accuracy", "Google Map Link"
  ];
  
  const colMap = {};
  
  for (let i = 0; i < requiredHeaders.length; i++) {
    const colName = requiredHeaders[i];
    const index = headers.indexOf(colName);
    if (index === -1) {
      throw new Error("Measurement sheet headers are malformed: missing " + colName);
    }
    colMap[colName] = index;
  }
  
  colMap["Measurement ID"] = headers.indexOf("Measurement ID");
  
  return colMap;
}

function getDeviceToken(deviceIdHash) {
  return CacheService.getScriptCache().get("token_" + deviceIdHash);
}

function validateDevice(deviceId, deviceToken, estate, operatorName) {
  const deviceIdHash = computeHash(deviceId);
  const tokenHash = computeHash(deviceToken);

  const appSheet = getOrCreateSheet("ApprovedDevices");
  const appData = appSheet.getDataRange().getValues();
  const headers = appData[0];
  
  if (!headers) return { success: false, errorType: "auth_failed", error: "No approved devices." };

  for (let i = 1; i < appData.length; i++) {
    const row = appData[i];
    if (row[headers.indexOf("deviceIdHash")] === deviceIdHash) {
      if (row[headers.indexOf("revoked")] === true) {
        return { success: false, errorType: "auth_failed", error: "Device revoked." };
      }
      
      const expiresAt = row[headers.indexOf("expiresAt")];
      if (expiresAt && new Date(expiresAt) < new Date()) {
        return { success: false, errorType: "subscription_expired", error: "Access expired. Please contact administrator." };
      }

      if (row[headers.indexOf("tokenHash")] !== tokenHash) {
         return { success: false, errorType: "auth_failed", error: "Invalid token." };
      }

      if (row[headers.indexOf("estate")] !== estate) {
         return { success: false, errorType: "auth_failed", error: "Estate mismatch." };
      }

      return { success: true };
    }
  }

  return { success: false, errorType: "auth_failed", error: "Device not approved." };
}

function updateLastSeen(deviceIdHash) {
  const appSheet = getOrCreateSheet("ApprovedDevices");
  const appData = appSheet.getDataRange().getValues();
  const headers = appData[0];
  
  if(!headers) return;

  for (let i = 1; i < appData.length; i++) {
    if (appData[i][headers.indexOf("deviceIdHash")] === deviceIdHash) {
      appSheet.getRange(i + 1, headers.indexOf("lastSeenAt") + 1).setValue(new Date().toISOString());
      appSheet.getRange(i + 1, headers.indexOf("lastSyncAt") + 1).setValue(new Date().toISOString());
      break;
    }
  }
}

/**
 * Send a Telegram notification to all configured chat IDs.
 * TELEGRAM_CHAT_ID supports a comma-separated list for multi-supervisor.
 */
function sendTelegramNotification_(text, inlineKeyboard) {
  if (!TELEGRAM_BOT_TOKEN || !TELEGRAM_CHAT_ID) {
    return { status: "failed", error: "Telegram not configured." };
  }

  const chatIds = String(TELEGRAM_CHAT_ID).split(',').map(s => s.trim()).filter(Boolean);
  if (chatIds.length === 0) {
    return { status: "failed", error: "No Telegram chat IDs configured." };
  }

  let allSucceeded = true;
  let lastError = "";

  chatIds.forEach(chatId => {
    try {
      const payload = {
        chat_id: chatId,
        text: text,
      };
      if (inlineKeyboard) {
        payload.reply_markup = { inline_keyboard: inlineKeyboard };
      }

      const response = UrlFetchApp.fetch(`https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage`, {
        method: "post",
        contentType: "application/json",
        muteHttpExceptions: true,
        payload: JSON.stringify(payload)
      });

      const responseCode = response.getResponseCode();
      const responseBody = response.getContentText();
      Logger.log(`Telegram [${chatId}] response: ${responseCode}`);

      if (responseCode !== 200) {
        allSucceeded = false;
        lastError = `Telegram Error [${chatId}]: ${responseCode} - ${responseBody}`;
      }
    } catch (e) {
      allSucceeded = false;
      lastError = `Telegram Exception [${chatId}]: ${e.toString()}`;
    }
  });

  return { status: allSucceeded ? "sent" : "partial", error: lastError };
}

function safeTelegramText_(value) {
  return String(value || "")
    .replace(/[\u0000-\u001F\u007F]/g, " ")
    .replace(/\s+/g, " ")
    .trim()
    .substring(0, 300);
}

function valueOrBlank_(val) {
  return (val === undefined || val === null) ? "" : val;
}

function createGoogleMapLink_(lat, lng) {
  if (lat && lng) {
    return `https://www.google.com/maps?q=${lat},${lng}`;
  }
  return "";
}

function resetMeasurementSheetHeaders() {
  // Runs the safe schema migration on all mapped estate sheets.
  // Idempotent — safe to run multiple times.
  for (const estateName in ESTATE_SHEET_MAP) {
    const sheetId = ESTATE_SHEET_MAP[estateName];
    if (sheetId) {
      try {
        const sheet = SpreadsheetApp.openById(sheetId).getSheets()[0];
        ensureMeasurementSheetSchema_(sheet);
      } catch (e) {
        Logger.log(`Error migrating schema for ${estateName}: ${e}`);
      }
    }
  }
}

function formatSriLankaDateTime_(timestamp) {
  const date = timestamp ? new Date(timestamp) : new Date();
  return Utilities.formatDate(date, "Asia/Colombo", "yyyy-MM-dd HH:mm:ss");
}

function handleAdminVerifyTotp(parsed) {
  const { code } = parsed;
  if (!code || !ADMIN_TOTP_SECRET) {
    return respondJson({ success: false, error: "Invalid authenticator code or backend not configured." });
  }

  // Basic rate limit via cache
  const cache = CacheService.getScriptCache();
  const attemptsKey = "admin_totp_attempts";
  let attempts = parseInt(cache.get(attemptsKey) || "0", 10);
  if (attempts >= 5) {
    return respondJson({ success: false, error: "Too many failed attempts. Try again later." });
  }

  if (verifyTotp_(code, ADMIN_TOTP_SECRET)) {
    cache.remove(attemptsKey);
    const sessionToken = "ADM-" + Utilities.getUuid();
    const tokenHash = computeHash(sessionToken + ADMIN_SESSION_SECRET);
    cache.put("admin_session_" + tokenHash, "valid", 1800); // 30 mins
    
    return respondJson({
      success: true,
      adminSessionToken: sessionToken,
      expiresAt: new Date(Date.now() + 1800000).toISOString()
    });
  } else {
    attempts++;
    cache.put(attemptsKey, attempts.toString(), 300); // lock for 5 mins
    return respondJson({ success: false, error: "Invalid authenticator code." });
  }
}

  });

function handleAdminFetchMeasurements(parsed) {
  if (!validateAdminSession_(parsed.adminSessionToken)) {
    return respondJson({ success: false, error: "Invalid or expired admin session." });
  }

  const { estate, division, fieldNo, dateFrom, dateTo, status } = parsed;
  if (!estate) return respondJson({ success: false, error: "Estate is required." });

  const sheetId = ESTATE_SHEET_MAP[estate];
  if (!sheetId) return respondJson({ success: false, error: "Unknown estate." });

  try {
    const sheet = SpreadsheetApp.openById(sheetId).getSheets()[0];
    const data = sheet.getDataRange().getValues();
    const headers = data[0];

    let colMap;
    try {
      colMap = getRequiredMeasurementColumnMap_(headers);
    } catch (e) {
      return respondJson({ success: false, error: e.message });
    }

    let measurements = [];
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const mDateStr = String(row[colMap["Date"]]);
      const mDate = new Date(mDateStr);
      
      // Division filter
      if (division && String(row[colMap["Division"]]).toLowerCase() !== division.toLowerCase()) continue;
      // FieldNo filter
      if (fieldNo && String(row[colMap["Field No"]]).toLowerCase() !== fieldNo.toLowerCase()) continue;
      
      // Date filter
      if (dateFrom && mDate < new Date(dateFrom)) continue;
      if (dateTo) {
        const to = new Date(dateTo);
        to.setHours(23, 59, 59, 999);
        if (mDate > to) continue;
      }

      // Status filter
      const recStatus = String(row[colMap["Tapping Recommendation"]]).toLowerCase();
      const abnFlag = String(row[colMap["Abnormal Flag"]]).toLowerCase() === 'yes';
      
      if (status === 'tappable' && !recStatus.includes('tappable')) continue;
      if (status === 'approaching' && !recStatus.includes('approaching')) continue;
      if (status === 'below' && !recStatus.includes('not ready') && !recStatus.includes('below')) continue;
      if (status === 'abnormal' && !abnFlag) continue;

      const midIdx = colMap["Measurement ID"];
      measurements.push({
        estate: row[colMap["Estate"]],
        division: row[colMap["Division"]],
        fieldNo: row[colMap["Field No"]],
        extent: row[colMap["Extent (Ha)"]],
        treeNo: row[colMap["Tree Number"]],
        caliperReading: row[colMap["Caliper Reading"]],
        girth: row[colMap["Girth"]],
        date: mDateStr,
        recommendationText: row[colMap["Tapping Recommendation"]],
        abnormalFlag: abnFlag,
        abnormalReason: row[colMap["Abnormal Reason"]],
        latitude: row[colMap["Latitude"]],
        longitude: row[colMap["Longitude"]],
        gpsAccuracy: row[colMap["GPS Accuracy"]],
        googleMapLink: row[colMap["Google Map Link"]],
        // Include measurementId if column exists; blank for legacy rows
        measurementId: midIdx !== -1 ? (row[midIdx] || "") : ""
      });
    }

    return respondJson({ success: true, measurements: measurements });
  } catch (e) {
    return respondJson({ success: false, error: "Error accessing sheet: " + e.toString() });
  }
}

function validateAdminSession_(token) {
  if (!token) return false;
  const tokenHash = computeHash(token + ADMIN_SESSION_SECRET);
  return CacheService.getScriptCache().get("admin_session_" + tokenHash) === "valid";
}

function handleAdminListDevices(parsed) {
  if (!validateAdminSession_(parsed.adminSessionToken)) {
    return respondJson({ success: false, error: "Invalid or expired admin session." });
  }

  const sheet = getOrCreateSheet("ApprovedDevices");
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return respondJson({ success: true, devices: [] });
  }

  const headers = data[0];
  const devices = [];
  for (let i = 1; i < data.length; i++) {
    const obj = {};
    headers.forEach((h, idx) => {
      // Omit sensitive hash fields from the response
      if (h !== "tokenHash") {
        obj[h] = data[i][idx];
      }
    });
    devices.push(obj);
  }

  return respondJson({ success: true, devices: devices });
}

function handleAdminRevokeDevice(parsed) {
  if (!validateAdminSession_(parsed.adminSessionToken)) {
    return respondJson({ success: false, error: "Invalid or expired admin session." });
  }

  const { deviceIdHash } = parsed;
  if (!deviceIdHash) {
    return respondJson({ success: false, error: "Device ID hash is required." });
  }

  const sheet = getOrCreateSheet("ApprovedDevices");
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const hashCol = headers.indexOf("deviceIdHash");
  const revokedCol = headers.indexOf("revoked");

  if (hashCol === -1 || revokedCol === -1) {
    return respondJson({ success: false, error: "ApprovedDevices sheet is missing required columns." });
  }

  for (let i = 1; i < data.length; i++) {
    if (data[i][hashCol] === deviceIdHash) {
      sheet.getRange(i + 1, revokedCol + 1).setValue(true);
      return respondJson({ success: true, message: "Device revoked." });
    }
  }

  return respondJson({ success: false, error: "Device not found." });
}

// ----------------------------------------------------
// TOTP Utilities
// ----------------------------------------------------
function verifyTotp_(code, secretBase32) {
  if (!code || code.length !== 6) return false;
  const keyBytes = base32Decode_(secretBase32);
  const timeStep = Math.floor(Date.now() / 30000);
  
  // Allow clock drift +/- 1 step
  for (let i = -1; i <= 1; i++) {
    const generated = generateTotp_(keyBytes, timeStep + i);
    if (constantTimeEqual_(code, generated)) {
      return true;
    }
  }
  return false;
}

function generateTotp_(keyBytes, counter) {
  // Convert counter to 8-byte array
  let counterBytes = [];
  for (let i = 7; i >= 0; i--) {
    counterBytes[i] = counter & 0xff;
    counter = Math.floor(counter / 256);
  }
  
  const mac = Utilities.computeHmacSignature(Utilities.MacAlgorithm.HMAC_SHA_1, counterBytes, keyBytes);
  const offset = mac[mac.length - 1] & 0x0f;
  const binary =
    ((mac[offset] & 0x7f) << 24) |
    ((mac[offset + 1] & 0xff) << 16) |
    ((mac[offset + 2] & 0xff) << 8) |
    (mac[offset + 3] & 0xff);
    
  let otp = (binary % 1000000).toString();
  while (otp.length < 6) {
    otp = '0' + otp;
  }
  return otp;
}

function base32Decode_(base32) {
  const alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';
  base32 = base32.toUpperCase().replace(/=+$/, '');
  let bits = '';
  let decoded = [];
  for (let i = 0; i < base32.length; i++) {
    let val = alphabet.indexOf(base32.charAt(i));
    if (val === -1) throw new Error("Invalid base32 character.");
    let bin = val.toString(2);
    while (bin.length < 5) bin = '0' + bin;
    bits += bin;
  }
  for (let i = 0; i + 8 <= bits.length; i += 8) {
    decoded.push(parseInt(bits.substr(i, 8), 2));
  }
  return decoded;
}

function constantTimeEqual_(a, b) {
  if (a.length !== b.length) return false;
  let result = 0;
  for (let i = 0; i < a.length; i++) {
    result |= a.charCodeAt(i) ^ b.charCodeAt(i);
  }
  return result === 0;
}
